cmake_minimum_required(VERSION 3.20)
option(DCR_ESP32 "Build with ESP-IDF (set IDF_PATH first)" OFF)
option(DCR_SENSOR_DEMO "Select SHT45/BMP280 demo instead of software fanout" OFF)
option(DCR_CONTROL "Build the bounded DCR control adapter" OFF)
if(DCR_ESP32)
  set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/platform/esp32")
  include($ENV{IDF_PATH}/tools/cmake/project.cmake)
  project(dcr_firmware)
else()
  project(dcr VERSION 0.1.0 LANGUAGES C)
  set(CMAKE_C_STANDARD 11)
  set(CMAKE_C_STANDARD_REQUIRED ON)
  option(DCR_SANITIZE "Enable host address sanitizer" OFF)
  if(DCR_SANITIZE)
    if(MSVC)
      add_compile_options(/fsanitize=address /Zi)
      add_link_options(/INCREMENTAL:NO)
    else()
      add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
      add_link_options(-fsanitize=address,undefined)
    endif()
  endif()
  include(CTest)
  add_library(cjson STATIC third_party/cjson/cJSON.c)
  target_include_directories(cjson PUBLIC third_party/cjson)
  add_library(dcr STATIC dcr/src/runtime.c dcr/src/lifecycle.c dcr/src/port.c dcr/src/bus.c dcr/src/safety.c
    dcr/src/validator.c dcr/src/config.c components/common/software.c
    components/control/control.c components/sensors/environment.c
    components/sensors/i2c_bus.c components/sensors/sensor_components.c
    platform/host/pal.c platform/host/i2c_bus.c platform/common/file.c)
  target_include_directories(dcr PUBLIC dcr/include platform/include components/common
    components/control components/sensors)
  target_link_libraries(dcr PUBLIC cjson)
  if(MSVC)
    target_compile_options(dcr PRIVATE /W4 /WX)
    target_compile_definitions(dcr PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_definitions(cjson PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(dcr PUBLIC Threads::Threads m)
    target_compile_options(dcr PRIVATE -Wall -Wextra -Werror -Wpedantic)
  endif()
  add_executable(dcr_demo main/host_main.c)
  target_link_libraries(dcr_demo PRIVATE dcr)
  add_executable(dcr_tests tests/test_dcr.c)
  target_link_libraries(dcr_tests PRIVATE dcr)
  add_test(NAME dcr_core COMMAND dcr_tests)
  add_executable(dcr_control_demo main/control_host_main.c)
  target_link_libraries(dcr_control_demo PRIVATE dcr)
  add_executable(dcr_control_tests tests/test_control.c)
  target_link_libraries(dcr_control_tests PRIVATE dcr)
  add_test(NAME dcr_control COMMAND dcr_control_tests)
  add_executable(dcr_bus_tests tests/test_bus.c)
  target_link_libraries(dcr_bus_tests PRIVATE dcr)
  add_test(NAME dcr_bus COMMAND dcr_bus_tests)
  add_executable(dcr_safety_tests tests/test_safety.c)
  target_link_libraries(dcr_safety_tests PRIVATE dcr)
  add_test(NAME dcr_safety COMMAND dcr_safety_tests)
  set_tests_properties(dcr_safety PROPERTIES TIMEOUT 30)
  add_executable(dcr_allocation_tests tests/test_allocation.c)
  target_link_libraries(dcr_allocation_tests PRIVATE dcr)
  add_test(NAME dcr_allocation COMMAND dcr_allocation_tests)
  set_tests_properties(dcr_allocation PROPERTIES TIMEOUT 30)
  add_test(NAME dcr_demo COMMAND dcr_demo ${CMAKE_CURRENT_SOURCE_DIR}/examples/fanout.json 600)
  add_test(NAME dcr_locked_demo COMMAND dcr_demo ${CMAKE_CURRENT_SOURCE_DIR}/examples/locked_fanout.json 600)
  set_tests_properties(dcr_locked_demo PROPERTIES TIMEOUT 30)
  add_executable(sensor_tests tests/test_sensors.c components/sensors/environment.c)
  target_include_directories(sensor_tests PRIVATE components/sensors)
  if(MSVC)
    target_compile_options(sensor_tests PRIVATE /W4 /WX)
  else()
    target_link_libraries(sensor_tests PRIVATE m)
    target_compile_options(sensor_tests PRIVATE -Wall -Wextra -Werror -Wpedantic)
  endif()
  add_test(NAME sensor_protocols COMMAND sensor_tests)
  set_tests_properties(sensor_protocols PROPERTIES TIMEOUT 10)
  set_tests_properties(dcr_core dcr_demo dcr_control dcr_bus PROPERTIES TIMEOUT 30)
endif()
