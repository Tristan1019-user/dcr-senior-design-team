set(DCR_MAIN_REQUIRES esp32 spiffs)
set(DCR_MAIN_INCLUDE_DIRS ".")
if(DCR_CONTROL)
  list(APPEND DCR_MAIN_REQUIRES esp_driver_uart)
  list(APPEND DCR_MAIN_INCLUDE_DIRS "../components/control" "../components/sensors")
endif()
if(DCR_SENSOR_DEMO)
  list(APPEND DCR_MAIN_INCLUDE_DIRS "../components/sensors")
endif()
idf_component_register(SRCS "esp_main.c" INCLUDE_DIRS ${DCR_MAIN_INCLUDE_DIRS} REQUIRES ${DCR_MAIN_REQUIRES})
if(DCR_SENSOR_DEMO)
  set(DCR_SENSOR_SECONDS 60 CACHE STRING "Bounded sensor demo duration")
  if(NOT DCR_SENSOR_SECONDS MATCHES "^[0-9]+$" OR DCR_SENSOR_SECONDS LESS 60 OR DCR_SENSOR_SECONDS GREATER 3600)
    message(FATAL_ERROR "DCR_SENSOR_SECONDS must be 60..3600")
  endif()
  math(EXPR sensor_duration_ms "${DCR_SENSOR_SECONDS} * 1000")
  target_compile_definitions(${COMPONENT_LIB} PRIVATE DCR_SENSOR_DEMO=1 DCR_SENSOR_DURATION_MS=${sensor_duration_ms})
endif()
if(DCR_CONTROL)
  target_compile_definitions(${COMPONENT_LIB} PRIVATE DCR_CONTROL=1)
endif()
spiffs_create_partition_image(storage "${PROJECT_DIR}/examples" FLASH_IN_PROJECT)
